const transporter = nodemailer.createTransport({
  service: 'gmail',
  auth: {
    user: process.env.GMAIL_USER,
    pass: process.env.GMAIL_PASS,
  },
});

const BASE_URL = process.env.NEXTAUTH_URL || 'https://zaydcluster.com';

interface EmailOptions {
  to: string;
  subject: string;
  html: string;
}

export async function sendEmail({ to, subject, html }: EmailOptions) {
  try {
    const info = await transporter.sendMail({
      from: `"ZaydCluster" <${process.env.GMAIL_USER}>`,
      to,
      subject,
      html,
    });
    console.log('Email sent:', info.messageId);
    return { success: true, messageId: info.messageId };
  } catch (error) {
    console.error('Email error:', error);
    return { success: false, error };
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { sendEmail, paymentConfirmationEmail, hostingReadyEmail } from '@/lib/email';
import { verifyXenditWebhook } from '@/lib/payment';

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const signature = request.headers.get('x-callback-token') || '';

    if (!verifyXenditWebhook(body, signature)) {
      console.error('Invalid Xendit webhook signature');
      return NextResponse.json({ success: false, message: 'Invalid signature' }, { status: 403 });
    }

    if (body.status === 'PAID') {
      const invoice = await db.invoice.findFirst({
        where: { invoiceNumber: body.external_id },
        include: { order: true, user: true },
      });

      if (invoice && invoice.status !== 'paid') {
        await db.invoice.update({
          where: { id: invoice.id },
          data: {
            status: 'paid',
            paidAt: new Date(),
            paymentMethod: body.payment_method || 'xendit',
            paymentRef: body.id,
          },
        });

        await db.order.update({
          where: { id: invoice.orderId },
          data: { status: 'active' },
        });

        const nextBilling = new Date();
        if (invoice.order.billingCycle === 'yearly') {
          nextBilling.setFullYear(nextBilling.getFullYear() + 1);
        } else {
          nextBilling.setMonth(nextBilling.getMonth() + 1);
        }

        const subscription = await db.subscription.create({
          data: {
            userId: invoice.userId,
            orderId: invoice.orderId,
            status: 'active',
            billingCycle: invoice.order.billingCycle,
            amount: invoice.totalAmount,
            nextBilling,
          },
        });

        // Send payment confirmation email
        await sendEmail({
          to: invoice.user.email,
          subject: `[ZaydCluster] Pembayaran Berhasil - ${invoice.invoiceNumber}`,
          html: paymentConfirmationEmail(invoice.user.name, invoice.invoiceNumber, invoice.totalAmount),
        });

        // ============ AUTO-PROVISION HOSTING ============
        try {
          const domain = invoice.order.domain || '';
          const subdomain = domain
            ? domain.replace(/\./g, '-')
            : invoice.user.name.toLowerCase()
                .replace(/[^a-z0-9]/g, '')
                .substring(0, 20) + '-' + invoice.orderId.substring(0, 6);

          // Call provision API on host (runs as root via systemd service)
          const provRes = await fetch('http://host.docker.internal:9999/provision', {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'Authorization': `Bearer ${process.env.PROVISION_SECRET || 'zc-prov-2026-secret'}`,
            },
            body: JSON.stringify({
              domain,
              subdomain,
              name: invoice.user.name,
              package: invoice.order.package || 'Starter',
              email: invoice.user.email,
              order: invoice.order.orderNumber,
            }),
            signal: AbortSignal.timeout(60_000),
          });
          const provData = await provRes.json();
          console.log('Provisioning result:', provData);

          if (provData.success) {
            const effectiveDomain = domain || `${subdomain}.pro99.my.id`;
            await sendEmail({
              to: invoice.user.email,
              subject: `[ZaydCluster] Hosting Anda Aktif - ${effectiveDomain}`,
              html: hostingReadyEmail(invoice.user.name, effectiveDomain, invoice.order.package, invoice.order.orderNumber),
            });
          }
        } catch (provisionError) {
          console.error('Auto-provision failed:', provisionError);
        }
        // ============ END AUTO-PROVISION ============
      }
    }

    return NextResponse.json({ success: true });
  } catch (error) {
    console.error('Xendit callback error:', error);
    return NextResponse.json({ success: false, message: 'Callback error' }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from "next/server";
import { db } from "@/lib/db";
import { z } from "zod";
import { generateOrderNumber, generateInvoiceNumber, PACKAGE_PRICES } from "@/lib/payment";
import { sendEmail, orderConfirmationEmail, invoiceEmail, passwordSetupEmail } from "@/lib/email";
import crypto from "crypto";

const orderSchema = z.object({
  name: z.string().min(2, "Nama minimal 2 karakter"),
  email: z.string().email("Email tidak valid"),
  phone: z.string().min(8, "Nomor telepon minimal 8 digit"),
  package: z.enum(["Starter", "Business", "Premium"]),
  billingCycle: z.enum(["monthly", "yearly"]).default("monthly"),
  domain: z.string().optional(),
  message: z.string().optional(),
});

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const validated = orderSchema.parse(body);

    const totalAmount = PACKAGE_PRICES[validated.package][validated.billingCycle];

    let user = await db.user.findUnique({ where: { email: validated.email } });
    let isNewUser = false;

    if (!user) {
      const bcrypt = await import("bcryptjs");
      const tempPassword = await bcrypt.hash("not-set-" + Date.now(), 12);
      user = await db.user.create({
        data: {
          name: validated.name,
          email: validated.email,
          phone: validated.phone,
          password: tempPassword,
          role: "client",
        },
      });
      isNewUser = true;
    }

    const orderNumber = generateOrderNumber();
    const invoiceNumber = generateInvoiceNumber();

    const order = await db.order.create({
      data: {
        orderNumber,
        userId: user.id,
        name: validated.name,
        email: validated.email,
        phone: validated.phone,
        package: validated.package,
        billingCycle: validated.billingCycle,
        domain: validated.domain || null,
        message: validated.message || null,
        status: "pending",
        totalAmount,
      },
    });

    const dueDate = new Date();
    dueDate.setDate(dueDate.getDate() + 3);

    const invoice = await db.invoice.create({
      data: {
        invoiceNumber,
        orderId: order.id,
        userId: user.id,
        amount: totalAmount,
        tax: 0,
        totalAmount,
        status: "unpaid",
        dueDate,
      },
    });

    const emailPromises = [];

    emailPromises.push(
      sendEmail({
        to: validated.email,
        subject: `[ZaydCluster] Pesanan Diterima - ${orderNumber}`,
        html: orderConfirmationEmail(validated.name, orderNumber, validated.package, validated.domain),
      })
    );

    emailPromises.push(
      sendEmail({
        to: validated.email,
        subject: `[ZaydCluster] Invoice #${invoiceNumber} - Rp ${totalAmount.toLocaleString("id-ID")}`,
        html: invoiceEmail(validated.name, invoiceNumber, totalAmount, dueDate.toLocaleDateString("id-ID")),
      })

    const callbackToken = request.headers.get('x-callback-token') || '';

    if (XENDIT_WEBHOOK_SECRET) {
      const expectedToken = crypto
        .createHmac('sha256', XENDIT_WEBHOOK_SECRET)
        .update(JSON.stringify(body))
        .digest('hex');

      if (callbackToken !== expectedToken) {
        return NextResponse.json({ success: false, message: 'Invalid token' }, { status: 403 });
      }
    }

    if (body.status === 'PAID') {
      const invoice = await db.invoice.findFirst({
        where: { invoiceNumber: body.external_id },
        include: { order: true, user: true },

        await db.invoice.update({
          where: { id: invoice.id },
          data: {
            status: 'paid',
            paidAt: new Date(),
            paymentMethod: body.payment_method || 'xendit',
            paymentRef: body.id,
          },
        });

        await db.order.update({
          where: { id: invoice.orderId },
          data: { status: 'active' },

        if (invoice.order.billingCycle === 'yearly') {

const XENDIT_SECRET_KEY = process.env.XENDIT_SECRET_KEY || '';
const XENDIT_WEBHOOK_SECRET = process.env.XENDIT_WEBHOOK_SECRET || '';
const BASE_URL = process.env.NEXTAUTH_URL || 'https://zaydcluster.com';

interface PaymentParams {
  orderId: string;
  invoiceNumber: string;
  amount: number;
  customerName: string;
  customerEmail: string;
  customerPhone: string;
  description?: string;
}

export async function createPaymentLink(params: PaymentParams) {
  const auth = Buffer.from(`${XENDIT_SECRET_KEY}:`).toString('base64');

  const dueDate = new Date();
  dueDate.setDate(dueDate.getDate() + 3);

  const payload = {
    external_id: params.invoiceNumber,
    amount: params.amount,
    description: params.description || `Invoice ${params.invoiceNumber} - ZaydCluster Hosting`,
    invoice_duration: 72,
    customer: {
      given_names: params.customerName,
      email: params.customerEmail,
      mobile_number: params.customerPhone,
    },
    success_redirect_url: `${BASE_URL}/invoices`,
    failure_redirect_url: `${BASE_URL}/invoices`,
    currency: 'IDR',
    items: [
      {
        name: `Hosting - ${params.invoiceNumber}`,
        quantity: 1,
        price: params.amount,
      },
    ],
  };

  try {
    const response = await fetch('https://api.xendit.co/v2/invoices', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Basic ${auth}`,
      },
      body: JSON.stringify(payload),
    });

    const data = await response.json();
    if (data.id) {
      return { success: true, paymentUrl: data.invoice_url, invoiceId: data.id };
    }
    return { success: false, error: data.message || 'Xendit error' };
  } catch (error) {
    console.error('Xendit error:', error);
    return { success: false, error: 'Payment gateway error' };
  }
}

export function verifyXenditWebhook(body: any, signature: string): boolean {
  if (!XENDIT_WEBHOOK_SECRET) return true;

  const webhookToken = crypto
    .createHmac('sha256', XENDIT_WEBHOOK_SECRET)
    .update(body.id + body.external_id + body.amount)
    .digest('hex');

  return webhookToken === signature;
}
