function OrderDialog({ selectedPkg, onClose, onSuccess }: { selectedPkg: string; onClose: () => void; onSuccess: (data: { orderNumber: string; invoiceNumber: string; isNewUser: boolean; setupToken?: string | null }) => void }) {
  const [loading, setLoading] = useState(false);
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    phone: '',
    package: selectedPkg,
    billingCycle: 'monthly',
    domain: '',
    message: '',
  });

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    try {
      const res = await fetch('/api/order', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(formData),
      });
      const data = await res.json();
      if (data.success) {
        toast.success(data.message);
        onSuccess({
          orderNumber: data.orderNumber,
          invoiceNumber: data.invoiceNumber,
          isNewUser: data.isNewUser,
          setupToken: data.setupToken,
        });
        setFormData({ name: '', email: '', phone: '', package: selectedPkg, billingCycle: 'monthly', domain: '', message: '' });
      } else {
        toast.error(data.message);
      }
    } catch {
      toast.error('Terjadi kesalahan jaringan. Silakan coba lagi.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <motion.div
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0 }}
      className="fixed inset-0 z-[60] flex items-center justify-center p-4"
      onClick={onClose}
    >
      <div className="absolute inset-0 bg-black/60 backdrop-blur-sm" />
      <motion.div
        initial={{ scale: 0.95, opacity: 0 }}
        animate={{ scale: 1, opacity: 1 }}
        exit={{ scale: 0.95, opacity: 0 }}

            <div className="space-y-2">
