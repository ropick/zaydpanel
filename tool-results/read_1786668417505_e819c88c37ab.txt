   162→def _hash_password(password):
   163→    """Hash password with SHA-256 + salt."""
   164→    salt = "zaydpanel-salt-2026"
   165→    return hashlib.sha256(f"{salt}:{password}".encode()).hexdigest()

   172→def _jwt_encode(payload, expiry_hours=24):
   173→    """Simple JWT-like token (base64 JSON)."""
   174→    import time as _time
   175→    payload["exp"] = int(_time.time()) + (expiry_hours * 3600)
   176→    payload["iat"] = int(_time.time())
   177→    header = base64.urlsafe_b64encode(json.dumps({"alg": "HS256", "typ": "JWT"}).encode()).decode().rstrip("=")
   178→    body = base64.urlsafe_b64encode(json.dumps(payload).encode()).decode().rstrip("=")
   179→    sig = hashlib.sha256(f"{CONF['JWT_SECRET']}.{header}.{body}".encode()).hexdigest()[:32]
   180→    return f"{header}.{body}.{sig}"
   181→

   183→def _jwt_decode(token):
   184→    """Decode JWT-like token."""
   185→    try:
   186→        parts = token.split(".")
   187→        if len(parts) != 3:
   188→            return None
   189→        header, body, sig = parts
   190→        expected_sig = hashlib.sha256(f"{CONF['JWT_SECRET']}.{header}.{body}".encode()).hexdigest()[:32]
   191→        if sig != expected_sig:
   192→            return None
   193→        payload = json.loads(base64.urlsafe_b64decode(body + "=="))
   194→        import time as _time
   195→        if payload.get("exp", 0) < int(_time.time()):
   196→            return None
   197→        return payload
   198→    except:
   199→        return None

   202→def _get_user_from_token(handler):
   203→    """Extract user from X-User-Token header."""
   204→    token = handler.headers.get("X-User-Token", "")
   205→    if not token:
   206→        return None
   207→    payload = _jwt_decode(token)

   209→        return None
   210→    return payload.get("user")

   306→    try:
   307→        content = Path(conf_path).read_text()
   308→        has_proxy = "proxy_pass" in content
   309→        if has_proxy:
   310→            lines = content.split("\n")

   313→                    return False
   314→            return True

   320→def _get_php_fpm_pool_status(domain):
   321→    pool_file = "%s/%s.conf" % (CONF["PHP_FPM_CONF_DIR"], domain)
   322→    if os.path.exists(pool_file):
   323→        try:
   324→            out, _, _ = _run("pgrep -c php-fpm 2>/dev/null || echo 0", check=False)
   325→            if int(out) > 0:
   326→                return True
   327→        except:
   328→            pass
   329→    try:
   330→        out, _, _ = _run("systemctl is-active php-fpm 2>/dev/null || echo inactive", check=False)
   331→        if out == "active":
   332→            return True
   333→    except:
   334→        pass

   358→def _get_sites():
   359→    sites = []
   360→    nginx_dir = Path(CONF["NGINX_CONF_DIR"])
   361→    if nginx_dir.exists():
   362→        for f in sorted(nginx_dir.glob("*.conf")):
   363→            domain = f.stem
   364→            if domain.startswith(SYSTEM_CONF_PREFIXES):
   365→                continue
   366→            try:
   367→                content = f.read_text()
   368→            except:
   369→                continue
   370→            home_dir = _site_home(domain)
   371→            has_home = os.path.isdir(home_dir)
   372→            has_ssl = "ssl_certificate" in content
   373→            has_wp = os.path.exists("%s/wp-login.php" % home_dir) if has_home else False
   374→            has_index = os.path.exists("%s/index.php" % home_dir) or os.path.exists("%s/index.html" % home_dir) if has_home else False
   375→            is_proxy = _is_proxy_config(str(f))
   376→            php_ver = _get_site_php_version(domain) if has_home and not is_proxy else None
   377→            php_fpm = _get_php_fpm_pool_status(domain) if has_home and not is_proxy else False
   378→            site_type = "proxy" if is_proxy else ("wordpress" if has_wp else ("active" if has_home and has_index else "empty"))
   379→            sites.append({
   380→                "domain": domain, "home": home_dir, "exists": has_home,
   381→                "ssl": has_ssl, "has_wp": has_wp, "php_version": php_ver,
   382→                "type": site_type, "php_fpm": php_fpm,
   383→            })
   384→    return sites

   399→def _fmt_net(bytes_val):
   400→    bytes_val = int(bytes_val)
   401→    if bytes_val >= 1073741824:
   402→        return "%.1f GB" % (bytes_val / 1073741824)
   403→    if bytes_val >= 1048576:
   404→        return "%.1f MB" % (bytes_val / 1048576)
   405→    if bytes_val >= 1024:
   406→        return "%.1f KB" % (bytes_val / 1024)
   407→    return "%d B" % bytes_val

   678→def handle_get_quota(handler):
   679→    """Get quota usage for current user."""
   680→    user = _get_request_user(handler)
   681→    if not user:
   682→        # If no user, return server-wide stats
   683→        return _ok(handler, {"disk_used": "0", "disk_limit": "Unlimited", "sites_used": 0, "sites_limit": 0,
   684→                              "databases_used": 0, "databases_limit": 0, "bandwidth_used": "0", "bandwidth_limit": "Unlimited"})
   685→    conn = _get_db()
   686→    u = conn.execute("SELECT package_id FROM users WHERE id=?", (user["id"],)).fetchone()
   687→    pkg = conn.execute("SELECT * FROM packages WHERE id=?", (u["package_id"],)).fetchone() if u["package_id"] else None
   688→    # Count user's resources
   689→    site_domains = [r["domain"] for r in conn.execute("SELECT domain FROM site_owners WHERE user_id=?", (user["id"],)).fetchall()]
   690→    sites_count = len(site_domains)
   691→    # Count databases for user's sites
   692→    db_count = 0
   693→    disk_used_mb = 0
   694→    for domain in site_domains:
   695→        home = _site_home(domain)
   696→        if os.path.isdir(home):
   697→            try:
   698→                out, _, _ = _run(f"du -sm {home} 2>/dev/null | cut -f1", check=False)
   699→                disk_used_mb += int(out) if out.isdigit() else 0
   700→            except:
   701→                pass
   702→        # Count databases
   703→        try:
   704→            db_name = domain.replace(".", "_")[:16]
   705→            out, _, _ = _run(f"mysql -N -e \"SELECT COUNT(*) FROM information_schema.schemata WHERE schema_name LIKE '{db_name}%'\"", check=False)
   706→            db_count += int(out) if out.strip().isdigit() else 0
   707→        except:
   708→            pass
   709→    # Email count
   710→    email_count = conn.execute("SELECT COUNT(*) as c FROM email_accounts WHERE user_id=?", (user["id"],)).fetchone()["c"]
   711→    # FTP count
   712→    ftp_count = conn.execute("SELECT COUNT(*) as c FROM ftp_accounts WHERE user_id=?", (user["id"],)).fetchone()["c"]
   713→    conn.close()
   714→    disk_used_str = f"{disk_used_mb}M" if disk_used_mb < 1024 else f"{disk_used_mb/1024:.1f}G"
   715→    result = {
   716→        "disk_used": disk_used_str,
   717→        "disk_limit": pkg["disk_quota"] if pkg else "Unlimited",
   718→        "sites_used": sites_count,
   719→        "sites_limit": pkg["max_sites"] if pkg else 0,
   720→        "databases_used": db_count,
   721→        "databases_limit": pkg["max_databases"] if pkg else 0,
   722→        "email_used": email_count,
   723→        "email_limit": pkg["max_email"] if pkg else 0,
   724→        "ftp_used": ftp_count,
   725→        "ftp_limit": pkg["max_ftp"] if pkg else 0,
   726→        "bandwidth_used": "0",
   727→        "bandwidth_limit": pkg["bandwidth"] if pkg else "Unlimited",
   728→    }
   729→    _ok(handler, result)

   732→def _check_quota(user_id, resource_type="sites"):
   733→    """Check if user can create more of a resource. Returns (allowed, limit, current)."""
   734→    conn = _get_db()
   735→    u = conn.execute("SELECT package_id FROM users WHERE id=?", (user_id,)).fetchone()
   736→    pkg = conn.execute("SELECT * FROM packages WHERE id=?", (u["package_id"],)).fetchone() if u["package_id"] else None
   737→    if not pkg:
   738→        conn.close()
   739→        return True, 0, 0
   740→    if resource_type == "sites":
   741→        current = conn.execute("SELECT COUNT(*) as c FROM site_owners WHERE user_id=?", (user_id,)).fetchone()["c"]
   742→        limit = pkg["max_sites"]
   743→    elif resource_type == "databases":
   744→        current = 0
   745→        limit = pkg["max_databases"]
   746→    elif resource_type == "email":
   747→        current = conn.execute("SELECT COUNT(*) as c FROM email_accounts WHERE user_id=?", (user_id,)).fetchone()["c"]
   748→        limit = pkg["max_email"]
   749→    elif resource_type == "ftp":
   750→        current = conn.execute("SELECT COUNT(*) as c FROM ftp_accounts WHERE user_id=?", (user_id,)).fetchone()["c"]
   751→        limit = pkg["max_ftp"]
   752→    else:
   753→        current, limit = 0, 0
   754→    conn.close()
   755→    if limit <= 0:
   756→        return True, limit, current
   757→    return current < limit, limit, current

   760→def handle_get_statistics(handler):
   761→    """Get bandwidth/disk statistics per domain."""
   762→    if not _is_admin(handler):
   763→        return _error(handler, "Admin only", 403)
   764→    stats = []
   765→    sites = _get_sites()
   766→    for site in sites:
   767→        home = _site_home(site["domain"])
   768→        disk_size = 0
   769→        if os.path.isdir(home):
   770→            try:
   771→                out, _, _ = _run(f"du -sm {home} 2>/dev/null | cut -f1", check=False)
   772→                disk_size = int(out) if out.strip().isdigit() else 0
   773→            except:
   774→                pass
   775→        # Try to get bandwidth from nginx logs
   776→        bw_rx = 0
   777→        bw_tx = 0
   778→        today = datetime.date.today().isoformat()
   779→        access_log = "%s/%s.access.log" % (CONF["LOG_DIR"], site["domain"])
   780→        if os.path.exists(access_log):
   781→            try:
   782→                out, _, _ = _run(f"awk '{{rx+=$10; tx+=$11}} END {{print rx, tx}}' {access_log} 2>/dev/null", check=False)
   783→                parts = out.split()
   784→                if len(parts) >= 2:
   785→                    bw_rx = int(parts[0])
   786→                    bw_tx = int(parts[1])
   787→            except:
   788→                pass
   789→        stats.append({
   790→            "domain": site["domain"],
   791→            "disk_mb": disk_size,
   792→            "disk_str": f"{disk_size}M" if disk_size < 1024 else f"{disk_size/1024:.1f}G",
   793→            "bandwidth_rx": _fmt_net(bw_rx),
   794→            "bandwidth_tx": _fmt_net(bw_tx),
   795→            "bandwidth_total": _fmt_net(bw_rx + bw_tx),
   796→            "date": today,
   797→        })
   798→    _ok(handler, stats)

  1006→def handle_webhook_receive(handler, data):
  1007→    """Receive billing webhook events from external billing systems."""
  1008→    event_type = data.get("event_type", "").strip()
  1009→    payload = json.dumps(data.get("payload", {}))
  1010→    if not event_type:
  1011→        return _error(handler, "event_type wajib")
  1012→    conn = _get_db()
  1013→    conn.execute("INSERT INTO webhook_events (event_type, payload, status) VALUES (?,?,?)",
  1014→                 (event_type, payload, "received"))
  1015→    conn.commit()
  1016→    event_id = conn.execute("SELECT last_insert_rowid()").fetchone()[0]
  1017→    # Process event
  1018→    try:
  1019→        if event_type == "payment_confirmed":
  1020→            username = data.get("payload", {}).get("username", "")
  1021→            pkg_slug = data.get("payload", {}).get("package_slug", "")
  1022→            if username and pkg_slug:
  1023→                pkg = conn.execute("SELECT id FROM packages WHERE slug=?", (pkg_slug,)).fetchone()
  1024→                if pkg:
  1025→                    conn.execute("UPDATE users SET status='active', package_id=? WHERE username=?", (pkg["id"], username))
  1026→                    conn.commit()
  1027→        elif event_type == "payment_expired":
  1028→            username = data.get("payload", {}).get("username", "")
  1029→            if username:
  1030→                conn.execute("UPDATE users SET status='suspended' WHERE username=? AND role='customer'", (username,))
  1031→                conn.commit()
  1032→        elif event_type == "user_created":
  1033→            username = data.get("payload", {}).get("username", "")
  1034→            password = data.get("payload", {}).get("password", _gen_password(12))
  1035→            email = data.get("payload", {}).get("email", "")
  1036→            pkg_slug = data.get("payload", {}).get("package_slug", "")
  1037→            if username:
  1038→                existing = conn.execute("SELECT id FROM users WHERE username=?", (username,)).fetchone()
  1039→                if not existing:
  1040→                    pkg = conn.execute("SELECT id FROM packages WHERE slug=?", (pkg_slug,)).fetchone() if pkg_slug else None
  1041→                    conn.execute("INSERT INTO users (username, password_hash, email, role, package_id) VALUES (?,?,?,?,?)",
  1042→                                 (username, _hash_password(password), email, "customer", pkg["id"] if pkg else None))
  1043→                    conn.commit()
  1044→        conn.execute("UPDATE webhook_events SET status='processed' WHERE id=?", (event_id,))
  1045→        conn.commit()
  1046→    except Exception as e:
  1047→        conn.execute("UPDATE webhook_events SET status='failed' WHERE id=?", (event_id,))
  1048→        conn.commit()
  1049→    conn.close()
  1050→    _ok(handler, {"id": event_id, "event_type": event_type, "status": "processed", "message": "Webhook processed"})
