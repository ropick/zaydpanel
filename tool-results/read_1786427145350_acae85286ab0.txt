   294→import { NextRequest, NextResponse } from 'next/server';
   295→import { db } from '@/lib/db';
   296→import { sendEmail, paymentConfirmationEmail, hostingReadyEmail } from '@/lib/email';
   297→import { verifyXenditWebhook } from '@/lib/payment';
   298→
   299→export async function POST(request: NextRequest) {
   300→  try {
   301→    const body = await request.json();
   302→    const signature = request.headers.get('x-callback-token') || '';
   303→
   304→    if (!verifyXenditWebhook(body, signature)) {
   305→      console.error('Invalid Xendit webhook signature');
   306→      return NextResponse.json({ success: false, message: 'Invalid signature' }, { status: 403 });
   307→    }
   308→
   309→    if (body.status === 'PAID') {
   310→      const invoice = await db.invoice.findFirst({
   311→        where: { invoiceNumber: body.external_id },
   312→        include: { order: true, user: true },
   313→      });
   314→
   315→      if (invoice && invoice.status !== 'paid') {
   316→        await db.invoice.update({
   317→          where: { id: invoice.id },
   318→          data: {
   319→            status: 'paid',
   320→            paidAt: new Date(),
   321→            paymentMethod: body.payment_method || 'xendit',
   322→            paymentRef: body.id,
   323→          },
   324→        });
   325→
   326→        await db.order.update({
   327→          where: { id: invoice.orderId },
   328→          data: { status: 'active' },
   329→        });
   330→
   331→        const nextBilling = new Date();
   332→        if (invoice.order.billingCycle === 'yearly') {
   333→          nextBilling.setFullYear(nextBilling.getFullYear() + 1);
   334→        } else {
   335→          nextBilling.setMonth(nextBilling.getMonth() + 1);
   336→        }
   337→
   338→        const subscription = await db.subscription.create({
   339→          data: {
   340→            userId: invoice.userId,
   341→            orderId: invoice.orderId,
   342→            status: 'active',
   343→            billingCycle: invoice.order.billingCycle,
   344→            amount: invoice.totalAmount,
   345→            nextBilling,
   346→          },
   347→        });
   348→
   349→        // Send payment confirmation email
   350→        await sendEmail({
   351→          to: invoice.user.email,
   352→          subject: `[ZaydCluster] Pembayaran Berhasil - ${invoice.invoiceNumber}`,
   353→          html: paymentConfirmationEmail(invoice.user.name, invoice.invoiceNumber, invoice.totalAmount),
   354→        });
   355→
   356→        // ============ AUTO-PROVISION HOSTING ============
   357→        try {
   358→          const domain = invoice.order.domain || '';
   359→          const subdomain = domain
   360→            ? domain.replace(/\./g, '-')
   361→            : invoice.user.name.toLowerCase()
   362→                .replace(/[^a-z0-9]/g, '')
   363→                .substring(0, 20) + '-' + invoice.orderId.substring(0, 6);
   364→
   365→          // Call provision API on host (runs as root via systemd service)
   366→          const provRes = await fetch('http://host.docker.internal:9999/provision', {
   367→            method: 'POST',
   368→            headers: {
   369→              'Content-Type': 'application/json',
   370→              'Authorization': `Bearer ${process.env.PROVISION_SECRET || 'zc-prov-2026-secret'}`,
   371→            },
   372→            body: JSON.stringify({
   373→              domain,
   374→              subdomain,
   375→              name: invoice.user.name,
   376→              package: invoice.order.package || 'Starter',
   377→              email: invoice.user.email,
   378→              order: invoice.order.orderNumber,
   379→            }),
   380→            signal: AbortSignal.timeout(60_000),
   381→          });
   382→          const provData = await provRes.json();
   383→          console.log('Provisioning result:', provData);
