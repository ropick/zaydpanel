   172→   172→def _jwt_encode(payload, expiry_hours=24):
   173→   173→    """Simple JWT-like token (base64 JSON)."""
   174→   174→    import time as _time
   175→   175→    payload["exp"] = int(_time.time()) + (expiry_hours * 3600)
   176→   176→    payload["iat"] = int(_time.time())
   177→   177→    header = base64.urlsafe_b64encode(json.dumps({"alg": "HS256", "typ": "JWT"}).encode()).decode().rstrip("=")
   178→   178→    body = base64.urlsafe_b64encode(json.dumps(payload).encode()).decode().rstrip("=")
   179→   179→    sig = hashlib.sha256(f"{CONF['JWT_SECRET']}.{header}.{body}".encode()).hexdigest()[:32]
   180→   180→    return f"{header}.{body}.{sig}"

   183→   183→def _jwt_decode(token):
   184→   184→    """Decode JWT-like token."""
   185→   185→    try:
   186→   186→        parts = token.split(".")
   187→   187→        if len(parts) != 3:
   188→   188→            return None
   189→   189→        header, body, sig = parts
   190→   190→        expected_sig = hashlib.sha256(f"{CONF['JWT_SECRET']}.{header}.{body}".encode()).hexdigest()[:32]
   191→   191→        if sig != expected_sig:
   192→   192→            return None
   193→   193→        payload = json.loads(base64.urlsafe_b64decode(body + "=="))
   194→   194→        import time as _time
   195→   195→        if payload.get("exp", 0) < int(_time.time()):
   196→   196→            return None
   197→   197→        return payload
   198→   198→    except:
   199→   199→        return None

   399→   399→def _fmt_net(bytes_val):
   400→   400→    bytes_val = int(bytes_val)
   401→   401→    if bytes_val >= 1073741824:
   402→   402→        return "%.1f GB" % (bytes_val / 1073741824)
   403→   403→    if bytes_val >= 1048576:
   404→   404→        return "%.1f MB" % (bytes_val / 1048576)
   405→   405→    if bytes_val >= 1024:
   406→   406→        return "%.1f KB" % (bytes_val / 1024)
   407→   407→    return "%d B" % bytes_val

   708→   708→            pass
